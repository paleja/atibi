# No rainbow screen, no blur: the box has no screen attached.
disable_splash=1

# Wired Ethernet only, so the wifi and bluetooth stacks have nothing to wait
# for at boot.
dtoverlay=disable-wifi
dtoverlay=disable-bt

# The boot order is fixed in the EEPROM, so skip the I2C read of it: about a
# second saved on a Pi 4. Remove this line if you ever need to net- or
# USB-boot this card.
force_eeprom_read=0
